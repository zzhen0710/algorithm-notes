## A. Are They the Same?

[题目](https://codeforces.com/contest/2280/problem/A)

```cpp
void solve() {
    int n, k;
    cin >> n >> k;

    int p = k & 1;
    int flex = 0, need = 0;

    for (int i = 0; i < n; ++i) {
        int x;
        cin >> x;
        if ((x & 1) != p) ++flex;
    }

    for (int i = 0; i < n; ++i) {
        int x;
        cin >> x;
        if ((x & 1) != p) ++need;
    }

    cout << (flex >= need ? "YES\n" : "NO\n");
}
```

加减 $2$ 能在同一奇偶类内调到任意目标值。操作 $x\mapsto 2x+k$ 只会把元素送到与 $k$ 同奇偶的一类，且送过去之后不能回到另一类。设 $p=k\bmod 2$，目标里与 $p$ 不同奇偶的数只能由同类源元素提供，其余源元素都能到达 $p$ 类。因此只需判断 `flex >= need`。时间 $O(n)$，额外空间 $O(1)$。

## B. Secret String Patterns

[题目](https://codeforces.com/contest/2280/problem/B)

```cpp
void solve() {
    int n;
    cin >> n;

    int first, last, total;

    cout << "? 1 1" << endl;
    cin >> first;

    cout << "? " << n << ' ' << n << endl;
    cin >> last;

    cout << "? 1 " << n << endl;
    cin >> total;

    if (total == n || total == 0) {
        cout << "! NO" << endl;
        return;
    }

    if (first == 0 || first == 1 && last == 1) {
        cout << "! YES" << endl;
        return;
    }

    cout << "? 1 " << total << endl;
    int pre;
    cin >> pre;

    if (pre == total)
        cout << "! NO" << endl;
    else
        cout << "! YES" << endl;
}
```

个人解用总数排除全同，再用首尾确定部分情况。剩下的情形只需验证前 `total` 位是否全为 `1`，最多四问。

### 官方思路：先刻画所有 NO 的串

```cpp
void solve() {
    int n;
    cin >> n;

    auto ask = [](int l, int r) {
        cout << "? " << l << ' ' << r << endl;
        int x;
        if (!(cin >> x) || x == -1) exit(0);
        return x;
    };

    int k = ask(1, n);
    bool found = k > 0 && k < n && ask(1, k) < k;
    cout << (found ? "! YES" : "! NO") << endl;
}
```

没有相邻 `01` 的串只能是若干个 `1` 接若干个 `0`。先查出 $k$ 个 `1`，就锁定了唯一可能的分界。前 $k$ 位若混进一个 `0`，后面就一定有一个 `1`，两者之间必有一次从 `0` 到 `1` 的相邻跳变。最多两问，查询后用 `endl` 刷新，读到 `-1` 立即退出。

## C. Reverse and Double

[题目](https://codeforces.com/contest/2280/problem/C)

```cpp
void solve() {
    int n;
    cin >> n;
    vector<ll> a(n + 1);
    for (int i = 1; i <= n; ++i)
        cin >> a[i];

    vector<ll> d(n);
    ll mx = -1;
    vector<int> pos;

    for (int i = 1; i < n; ++i) {
        d[i] = abs(a[i] - a[i + 1]);

        if (d[i] > mx) {
            mx = d[i];
            pos.clear();
            pos.push_back(i);
        } else if (d[i] == mx) {
            pos.push_back(i);
        }
    }

    vector<ll> pre(n + 1), suf(n + 1);
    for (int i = 1; i < n; ++i)
        pre[i] = max(pre[i - 1], d[i]);
    for (int i = n - 1; i >= 1; --i)
        suf[i] = max(suf[i + 1], d[i]);

    auto get_outside_max = [&](int l, int r) -> ll {
        ll res = 0;
        if (l - 2 >= 1) res = pre[l - 2];
        if (r + 1 <= n - 1) res = max(res, suf[r + 1]);
        return res;
    };

    ll ans = mx;

    if (pos.size() >= 3) {
    } else if (pos.size() == 2) {
        int p1 = pos[0], p2 = pos[1];

        ll C = 0;
        for (int i = p1 + 1; i <= p2 - 1; ++i)
            C = max(C, 2 * d[i]);

        ll L = abs(a[p1] - 2 * a[p2]);
        ll R = abs(a[p2 + 1] - 2 * a[p1 + 1]);
        ll Out = get_outside_max(p1 + 1, p2);

        ans = min(ans, max({C, L, R, Out}));
    } else if (pos.size() == 1) {
        int p = pos[0];

        ll C = 0;
        for (int r = p + 1; r <= n; ++r) {
            if (r > p + 1)
                C = max(C, 2 * d[r - 1]);

            ll L = abs(a[p] - 2 * a[r]);
            ll R = (r < n) ? abs(2 * a[p + 1] - a[r + 1]) : 0;
            ll Out = get_outside_max(p + 1, r);

            ans = min(ans, max({C, L, R, Out}));
        }

        C = 0;
        for (int l = p; l >= 1; --l) {
            if (l < p)
                C = max(C, 2 * d[l]);

            ll L = (l > 1) ? abs(a[l - 1] - 2 * a[p]) : 0;
            ll R = abs(2 * a[l] - a[p + 1]);
            ll Out = get_outside_max(l, p);

            ans = min(ans, max({C, L, R, Out}));
        }
    }

    cout << ans << '\n';
}
```

令 $d_i=\lvert a_i-a_{i+1}\rvert$，原最大值为 $M$。一个旧最大缝隙若在区间内部，差值会翻倍；若在外部，差值保留。要让答案严格小于 $M$，必须把每一处旧最大缝隙切在操作边界。

- **至少三处**：两个边界不够切，保留不操作的答案 $M$。
- **恰好两处**：若缝隙编号为 $p_1,p_2$，唯一候选是 $[p_1+1,p_2]$。
- **恰好一处**：固定它为左或右边界，向另一侧扫描并增量维护内部最大差值。

### 1-based 下标图

```text
                      L    inside                R
before: ... a[l-1]    |    a[l] ... a[r]         |    a[r+1] ...
after:  ... a[l-1]    |    2*a[r] ... 2*a[l]     |    a[r+1] ...
old gaps: 1..l-2      l-1  l..r-1                r    r+1..n-1
effect:   unchanged   new  doubled               new  unchanged
```

翻转后，内部差值顺序反转、数值翻倍，取原 $d_l,\ldots,d_{r-1}$ 的最大值再乘 $2$。边界重新计算 $\lvert a_{l-1}-2a_r\rvert$ 与 $\lvert 2a_l-a_{r+1}\rvert$。外部最大值由 `pre[l-2]` 和 `suf[r+1]` 取较大者。碰到数组端点时忽略不存在的边界，单元素区间则没有内部差值。

**官方的线段树怎么用？** 树上存原差值，翻转与乘二都通过上述公式计算。用半开区间分别查询内部和左右外部，单个候选花 $O(\log n)$，总计 $O(n\log n)$。个人解的外部固定贴两端，内部随扫描每次增加一项，用前后缀最大值和滚动最大值把总时间降到 $O(n)$。两者空间均为 $O(n)$。

## D. Balanced Subsets

[题目](https://codeforces.com/contest/2280/problem/D)

```cpp
constexpr ll MOD = 998244353;

void solve() {
    int n;
    cin >> n;

    vector<char> a(3 * n + 1);
    int mx = 0;

    for (int i = 0; i < n; ++i) {
        int x;
        cin >> x;
        a[x] = true;
        mx = max(mx, x);
    }

    ll dp[8][3] = {};
    dp[0][0] = 1;

    for (int x = mx; x >= 1; --x) {
        ll ndp[8][3] = {};

        for (int mask = 0; mask < 8; ++mask) {
            for (int cnt = 0; cnt < 3; ++cnt) {
                if (dp[mask][cnt] == 0)
                    continue;

                for (int u = 0; u <= a[x]; ++u) {
                    int nc = cnt;
                    if (u && __builtin_popcount(mask) != 2)
                        ++nc;
                    if (nc > 2)
                        continue;

                    int nm = ((mask << 1) & 7) | u;

                    ndp[nm][nc] = (ndp[nm][nc] + dp[mask][cnt]) % MOD;
                }
            }
        }

        memcpy(dp, ndp, sizeof(dp));
    }

    ll ans = 0;

    for (int mask = 0; mask < 8; ++mask)
        for (int cnt = 0; cnt < 3; ++cnt)
            ans = (ans + dp[mask][cnt]) % MOD;

    cout << (ans - 1 + MOD) % MOD << '\n';
}
```

**先确定右边三个值，再决定当前值。** 从大到小扫数值轴，`mask` 的低三位依次对应 $x+1,x+2,x+3$ 是否已选。选 $x$ 且右侧恰有两个 `1` 时稳定，否则这次选择会消耗一个不稳定名额。用 `cnt` 记已消耗的名额，超过两个就不转移。不存在的值只允许 `u=0`。进入下一轮时移位、截断并写入 `u`，最后从计数中减掉空集。值域上界是 $3n$，时间 $O(n)$，存在表占 $O(n)$，滚动状态只有 $8\times3$ 项。

### 官方思路：先刻画合法块，再计数

子集至少有三个元素时，最大的两个必然不稳定，其余元素都必须稳定。对任意一个稳定元素，后面第二个被选值距它至多 $3$，第三个被选值若存在则距它大于 $3$。因此相邻被选值的间距只能是 $1$ 或 $2$，不能连续出现两个间距 $2$，也不能连续出现三个间距 $1$。按间距 $2$ 分段，内部每段恰有两个或三个连续的 `1`，首尾还允许单个 `1`。给末端补一个 `0`，得到 `(10)?(110|1110)*(10)?` 的结构。

1. **先压缺口。** 原集合相邻值的差为 $1,2$ 或至少 $3$ 时，分别追加 `1`、`01`、`001`。两个连续的不可选位已经足以截断大子集的拼接。模式中的 `0` 表示不选，即使该位置本来可选也能放 `0`。
2. **按块转移。** `dp1` 表示从空模式开始，`dp2` 表示先接一个 `10`。两者都能继续追加 `110` 或 `1110`，分别从新块起点前一格接上。每个位置的 `dp1` 放一个空起点，汇总时扣掉它。再单独枚举结尾额外接一个 `10` 的方案。
3. **扣掉重复的小子集。** 先加 $n+\binom n2$，因为所有大小为 $1,2$ 的子集都合法。分块计数还会把单点算两次，把压缩后间距为 $1$ 或 $2$ 的点对各算一次，分别扣掉即可。官方写法也是 $O(n)$，核心收获是先证明形状，再把选或不选改写成拼块。
